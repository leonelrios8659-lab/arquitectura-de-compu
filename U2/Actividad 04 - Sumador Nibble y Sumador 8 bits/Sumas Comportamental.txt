module sumadorI (
    input  [3:0] A,
    input  [3:0] B,
    input  Cin,
    output [3:0] C,
    output Cout
);

    assign {Cout, C} = A + B + Cin;

endmodule


module sumadorII (input [7:0] a, input [7:0] b, input cin, output [8:0] c, output cout);
		
		wire carreo;
		sumadorI primero (.A(a[3:0]), .B(b[3:0]),.Cin (carreo), .C(c[3:0]), .Cout(carreo));
		sumadorI segundo (.A(a[7:4]), .B(b[7:4]),.Cin (cin), .C(c[7:4]), .Cout(cout));
		
		
endmodule


module tb_sum4b;

	reg [3:0] A;
	reg [3:0] B;
	reg Cin;
	wire [3:0] C;
	wire Cout;
	
	sumadorI u1 (.A(A), .B(B), .Cin (Cin), .C(C), .Cout(Cout));
		
		initial begin 
		#10 A=1; B=5; Cin=0;
		#10 A=3; B=5; Cin=0;
		#10 A=6; B=5; Cin=1;
		$finish; 
		end 
		
endmodule



module tb_sum_8b;

	reg [7:0] a;
	reg [7:0] b;
	reg cin;
	wire [7:0] c;
	wire cout;
	
	sumadorI u1 (.a(a), .b(b),.cin (cin), .c(c), .cout(cout));
		
		initial begin 
		a=0; b=0; cin=0; 
		#10 a=1; b=5; cin=0;
		#10 a=3; b=5; cin=0;
		#10 a=6; b=5; cin=1;
		$finish; 
		end 
		
endmodule
